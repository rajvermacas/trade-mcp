import pandas as pd
import numpy as np
import ta
from datetime import datetime, time
import warnings
warnings.filterwarnings('ignore')

class EnhancedSupertrendStrategy:
    """
    Enhanced Supertrend Strategy for Nifty Trading
    Combines multiple filters for improved performance
    """
    
    def __init__(self, capital=1000000, risk_per_trade=0.02):
        self.capital = capital
        self.risk_per_trade = risk_per_trade
        self.trades = []
        self.position = None
        
        # Strategy parameters
        self.atr_period = 10
        self.st_multiplier = 3.0
        self.rsi_period = 14
        self.volume_period = 20
        self.volume_threshold = 1.5
        
    def calculate_atr(self, df, period=10):
        """Calculate Average True Range"""
        return ta.volatility.AverageTrueRange(
            high=df['high'], 
            low=df['low'], 
            close=df['close'], 
            window=period
        ).average_true_range()
    
    def calculate_supertrend(self, df, atr_period=10, multiplier=3.0):
        """Calculate Supertrend indicator"""
        high = df['high']
        low = df['low']
        close = df['close']
        
        # Calculate ATR
        atr = self.calculate_atr(df, atr_period)
        
        # Calculate basic upper and lower bands
        hl2 = (high + low) / 2
        basic_upper = hl2 + (multiplier * atr)
        basic_lower = hl2 - (multiplier * atr)
        
        # Calculate final upper and lower bands
        final_upper = basic_upper.copy()
        final_lower = basic_lower.copy()
        
        for i in range(1, len(df)):
            # Final Upper Band
            if basic_upper.iloc[i] < final_upper.iloc[i-1] or close.iloc[i-1] > final_upper.iloc[i-1]:
                final_upper.iloc[i] = basic_upper.iloc[i]
            else:
                final_upper.iloc[i] = final_upper.iloc[i-1]
                
            # Final Lower Band    
            if basic_lower.iloc[i] > final_lower.iloc[i-1] or close.iloc[i-1] < final_lower.iloc[i-1]:
                final_lower.iloc[i] = basic_lower.iloc[i]
            else:
                final_lower.iloc[i] = final_lower.iloc[i-1]
        
        # Calculate Supertrend
        supertrend = pd.Series(index=df.index, dtype=float)
        trend = pd.Series(index=df.index, dtype=int)
        
        supertrend.iloc[0] = final_lower.iloc[0]
        trend.iloc[0] = 1
        
        for i in range(1, len(df)):
            if close.iloc[i] <= final_lower.iloc[i]:
                supertrend.iloc[i] = final_lower.iloc[i]
                trend.iloc[i] = 1
            elif close.iloc[i] >= final_upper.iloc[i]:
                supertrend.iloc[i] = final_upper.iloc[i]
                trend.iloc[i] = -1
            else:
                supertrend.iloc[i] = supertrend.iloc[i-1]
                trend.iloc[i] = trend.iloc[i-1]
                
        return supertrend, trend, atr
    
    def calculate_volume_filter(self, df):
        """Calculate volume confirmation filter"""
        volume_avg = df['volume'].rolling(window=self.volume_period).mean()
        volume_surge = df['volume'] > (volume_avg * self.volume_threshold)
        return volume_surge
    
    def calculate_rsi_filter(self, df):
        """Calculate RSI momentum filter"""
        rsi = ta.momentum.RSIIndicator(close=df['close'], window=self.rsi_period).rsi()
        
        # RSI conditions for entry
        rsi_buy_condition = (rsi >= 40) & (rsi <= 80)
        rsi_sell_condition = (rsi >= 20) & (rsi <= 60)
        
        return rsi, rsi_buy_condition, rsi_sell_condition
    
    def session_filter(self, timestamp):
        """Check if timestamp is within trading session"""
        if isinstance(timestamp, str):
            timestamp = pd.to_datetime(timestamp)
        
        trade_time = timestamp.time()
        
        # Primary session: 9:30 AM - 12:30 PM
        # Secondary session: 1:30 PM - 3:15 PM
        primary_session = (time(9, 30) <= trade_time <= time(12, 30))
        secondary_session = (time(13, 30) <= trade_time <= time(15, 15))
        
        return primary_session or secondary_session
    
    def calculate_position_size(self, price, atr, risk_amount):
        """Calculate dynamic position size based on ATR"""
        risk_per_share = atr * self.st_multiplier
        if risk_per_share == 0:
            return 0
        
        position_size = risk_amount / risk_per_share
        max_position_value = self.capital * 0.05  # Maximum 5% of capital
        max_shares = max_position_value / price
        
        return min(position_size, max_shares)
    
    def generate_signals(self, df_1h, df_4h):
        """
        Generate trading signals using enhanced Supertrend strategy
        """
        # Calculate indicators for both timeframes
        st_1h, trend_1h, atr_1h = self.calculate_supertrend(df_1h)
        st_4h, trend_4h, _ = self.calculate_supertrend(df_4h)
        
        # Calculate filters
        volume_surge = self.calculate_volume_filter(df_1h)
        rsi, rsi_buy_ok, rsi_sell_ok = self.calculate_rsi_filter(df_1h)
        
        # Align 4H trend with 1H data
        trend_4h_aligned = trend_4h.reindex(df_1h.index, method='ffill')
        
        signals = pd.DataFrame(index=df_1h.index)
        signals['price'] = df_1h['close']
        signals['supertrend_1h'] = st_1h
        signals['trend_1h'] = trend_1h
        signals['trend_4h'] = trend_4h_aligned
        signals['volume_surge'] = volume_surge
        signals['rsi'] = rsi
        signals['atr'] = atr_1h
        
        # Generate entry signals
        signals['buy_signal'] = (
            (trend_1h == 1) &  # 1H uptrend
            (trend_1h.shift(1) == -1) &  # Previous bar was downtrend (signal)
            (trend_4h_aligned == 1) &  # 4H uptrend
            volume_surge &  # Volume confirmation
            rsi_buy_ok &  # RSI not overbought
            [self.session_filter(ts) for ts in df_1h.index]  # Session filter
        )
        
        signals['sell_signal'] = (
            (trend_1h == -1) &  # 1H downtrend  
            (trend_1h.shift(1) == 1) &  # Previous bar was uptrend (signal)
            (trend_4h_aligned == -1) &  # 4H downtrend
            volume_surge &  # Volume confirmation
            rsi_sell_ok &  # RSI not oversold
            [self.session_filter(ts) for ts in df_1h.index]  # Session filter
        )
        
        # Calculate signal strength (ML scoring)
        signals['signal_strength'] = 0
        
        # Volume surge intensity (0-3 points)
        volume_intensity = df_1h['volume'] / df_1h['volume'].rolling(20).mean()
        signals['signal_strength'] += np.where(volume_intensity > 2.0, 3,
                                              np.where(volume_intensity > 1.8, 2,
                                                      np.where(volume_intensity > 1.5, 1, 0)))
        
        # RSI momentum (0-2 points)
        signals['signal_strength'] += np.where((rsi > 50) & (rsi < 70), 2,
                                              np.where((rsi > 30) & (rsi < 80), 1, 0))
        
        # Time of day factor (0-1 points)
        morning_boost = [1 if time(9, 30) <= pd.to_datetime(ts).time() <= time(11, 30) else 0 
                        for ts in df_1h.index]
        signals['signal_strength'] += morning_boost
        
        # Volatility factor (0-2 points)
        atr_avg = atr_1h.rolling(20).mean()
        signals['signal_strength'] += np.where(atr_1h < atr_avg * 1.2, 2,
                                              np.where(atr_1h < atr_avg * 1.5, 1, 0))
        
        return signals
    
    def backtest(self, df_1h, df_4h):
        """
        Run backtest on historical data
        """
        signals = self.generate_signals(df_1h, df_4h)
        
        self.trades = []
        self.position = None
        portfolio_value = self.capital
        
        for i, (timestamp, row) in enumerate(signals.iterrows()):
            current_price = row['price']
            
            # Exit conditions
            if self.position:
                exit_signal = False
                exit_reason = ""
                
                # Supertrend reversal
                if self.position['type'] == 'long' and row['trend_1h'] == -1:
                    exit_signal = True
                    exit_reason = "Supertrend reversal"
                elif self.position['type'] == 'short' and row['trend_1h'] == 1:
                    exit_signal = True
                    exit_reason = "Supertrend reversal"
                
                # 4H trend change
                elif self.position['4h_trend'] != row['trend_4h']:
                    exit_signal = True
                    exit_reason = "4H trend change"
                
                # Session end
                elif not self.session_filter(timestamp):
                    exit_signal = True
                    exit_reason = "Session end"
                
                if exit_signal:
                    # Calculate P&L
                    if self.position['type'] == 'long':
                        pnl = (current_price - self.position['entry_price']) * self.position['quantity']
                    else:
                        pnl = (self.position['entry_price'] - current_price) * self.position['quantity']
                    
                    portfolio_value += pnl
                    
                    # Record trade
                    trade = {
                        'entry_time': self.position['entry_time'],
                        'exit_time': timestamp,
                        'type': self.position['type'],
                        'entry_price': self.position['entry_price'],
                        'exit_price': current_price,
                        'quantity': self.position['quantity'],
                        'pnl': pnl,
                        'pnl_pct': (pnl / (self.position['entry_price'] * self.position['quantity'])) * 100,
                        'exit_reason': exit_reason
                    }
                    self.trades.append(trade)
                    self.position = None
            
            # Entry signals (only if no position)
            if not self.position:
                risk_amount = portfolio_value * self.risk_per_trade
                
                # Buy signal
                if row['buy_signal'] and row['signal_strength'] >= 6:
                    position_size = self.calculate_position_size(current_price, row['atr'], risk_amount)
                    if position_size > 0:
                        self.position = {
                            'type': 'long',
                            'entry_time': timestamp,
                            'entry_price': current_price,
                            'quantity': position_size,
                            '4h_trend': row['trend_4h'],
                            'signal_strength': row['signal_strength']
                        }
                
                # Sell signal
                elif row['sell_signal'] and row['signal_strength'] >= 6:
                    position_size = self.calculate_position_size(current_price, row['atr'], risk_amount)
                    if position_size > 0:
                        self.position = {
                            'type': 'short',
                            'entry_time': timestamp,
                            'entry_price': current_price,
                            'quantity': position_size,
                            '4h_trend': row['trend_4h'],
                            'signal_strength': row['signal_strength']
                        }
        
        return self.analyze_performance()
    
    def analyze_performance(self):
        """Analyze backtest performance"""
        if not self.trades:
            return {"error": "No trades generated"}
        
        trades_df = pd.DataFrame(self.trades)
        
        # Performance metrics
        total_trades = len(trades_df)
        winning_trades = len(trades_df[trades_df['pnl'] > 0])
        losing_trades = len(trades_df[trades_df['pnl'] < 0])
        win_rate = (winning_trades / total_trades) * 100 if total_trades > 0 else 0
        
        total_pnl = trades_df['pnl'].sum()
        total_return = (total_pnl / self.capital) * 100
        
        avg_win = trades_df[trades_df['pnl'] > 0]['pnl_pct'].mean() if winning_trades > 0 else 0
        avg_loss = trades_df[trades_df['pnl'] < 0]['pnl_pct'].mean() if losing_trades > 0 else 0
        
        # Calculate drawdown
        cumulative_returns = (1 + trades_df['pnl_pct'] / 100).cumprod()
        running_max = cumulative_returns.expanding().max()
        drawdown = (cumulative_returns - running_max) / running_max * 100
        max_drawdown = drawdown.min()
        
        # Sharpe ratio (simplified)
        returns_std = trades_df['pnl_pct'].std()
        sharpe_ratio = (total_return / returns_std) if returns_std > 0 else 0
        
        return {
            'total_return': total_return,
            'total_trades': total_trades,
            'winning_trades': winning_trades,
            'losing_trades': losing_trades,
            'win_rate': win_rate,
            'avg_win': avg_win,
            'avg_loss': avg_loss,
            'max_drawdown': max_drawdown,
            'sharpe_ratio': sharpe_ratio,
            'profit_factor': abs(avg_win * winning_trades / avg_loss / losing_trades) if avg_loss != 0 else 0,
            'trades': trades_df
        }
